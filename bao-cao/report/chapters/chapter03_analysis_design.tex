\chapter{Phân tích thiết kế hệ thống}

\section{Tổng quan hệ thống}

Hệ thống SmartNutri bao gồm ba thành phần chính:

\begin{enumerate}
    \item \textbf{Ứng dụng di động (Mobile App):} Ứng dụng Flutter chạy trên Android/iOS, là điểm tương tác chính với người dùng cuối.
    \item \textbf{Backend Firebase:} Bao gồm Authentication, Firestore, Storage và Cloud Functions, xử lý toàn bộ logic nghiệp vụ và lưu trữ dữ liệu.
    \item \textbf{Trang quản trị (Admin Web):} Ứng dụng React cho phép quản trị viên quản lý dữ liệu thực phẩm, người dùng và theo dõi hoạt động hệ thống.
\end{enumerate}

Các actor chính của hệ thống:
\begin{itemize}
    \item \textbf{Người dùng (User):} Người dùng cuối sử dụng ứng dụng di động để theo dõi dinh dưỡng.
    \item \textbf{Quản trị viên (Admin):} Người quản lý dữ liệu thực phẩm, danh mục và người dùng qua trang admin web.
    \item \textbf{Hệ thống AI (Gemini):} Dịch vụ AI bên ngoài hỗ trợ nhận diện món ăn và phân tích dinh dưỡng.
\end{itemize}

\subsection{Yêu cầu phi chức năng}

\subsubsection{Hiệu năng}
\begin{itemize}
    \item Ứng dụng khởi động và hiển thị màn hình chính trong thời gian chấp nhận được (mục tiêu dưới 3 giây trên thiết bị Android tầm trung).
    \item Truy vấn nhật ký ăn uống theo ngày hiển thị kết quả nhanh, tận dụng Firestore local cache cho dữ liệu đã từng truy cập.
    \item Thời gian quét món ăn bằng AI phụ thuộc vào độ trễ mạng và thời gian phản hồi của Gemini API; mục tiêu hiển thị kết quả trong vòng 10 giây kể từ khi gửi ảnh.
    \item Biểu đồ dinh dưỡng được tính toán và hiển thị mượt mà trên tập dữ liệu nhật ký theo tuần.
\end{itemize}

\subsubsection{Độ tin cậy và tính sẵn sàng}
\begin{itemize}
    \item Hệ thống backend phụ thuộc vào hạ tầng Firebase của Google; tính sẵn sàng của ứng dụng ở mức tương đương Firebase SLA.
    \item Dữ liệu người dùng được lưu trên Firestore với cơ chế sao lưu tự động của Firebase.
    \item Ứng dụng hỗ trợ chế độ offline cơ bản: dữ liệu đã tải được giữ trong Firestore local cache, cho phép người dùng xem nhật ký và danh sách thực phẩm khi không có kết nối mạng. Các thao tác ghi (thêm món, quét AI) yêu cầu kết nối mạng để đồng bộ.
\end{itemize}

\subsubsection{An ninh và bảo mật}
\begin{itemize}
    \item Xác thực người dùng qua Firebase Authentication, sử dụng token JWT để xác định danh tính trong mọi request đến Firestore và Cloud Functions.
    \item Firestore Security Rules đảm bảo mỗi người dùng chỉ có thể đọc và ghi dữ liệu thuộc về chính mình (phân tách theo UID).
    \item Toàn bộ kết nối giữa client và Firebase sử dụng SSL/TLS.
    \item Gemini API key được lưu dưới dạng biến môi trường trên Cloud Functions, không xuất hiện trong mã nguồn phía client, tránh nguy cơ bị trích xuất từ file APK.
    \item Trang admin yêu cầu tài khoản có admin claim mới có thể gọi các Cloud Function quản trị (setAdminRole, setUserSubscription).
\end{itemize}

\subsubsection{Khả bảo trì}
\begin{itemize}
    \item Code Flutter được tổ chức theo cấu trúc feature-first: mỗi tính năng nằm trong thư mục riêng, tách biệt giữa domain (model, service) và presentation (widget, page).
    \item Cloud Functions và Admin Web sử dụng TypeScript, cung cấp kiểm tra kiểu tĩnh, giảm thiểu lỗi runtime và hỗ trợ quá trình bảo trì.
    \item Các service class trong Flutter đảm nhận toàn bộ logic giao tiếp với Firebase, tránh việc gọi Firebase trực tiếp từ widget.
\end{itemize}

\subsubsection{Khả mở rộng}
\begin{itemize}
    \item Cấu trúc feature-first cho phép thêm tính năng mới bằng cách tạo module mới mà không ảnh hưởng đến các module hiện có.
    \item Firestore là NoSQL document database, không yêu cầu schema cố định, cho phép thêm trường mới vào document mà không cần migrate dữ liệu cũ.
    \item Cloud Functions có thể được thêm mới hoặc cập nhật độc lập mà không cần phát hành lại ứng dụng di động.
    \item Kiến trúc hiện tại đã có nền tảng để mở rộng sang iOS khi cần (dùng chung codebase Flutter).
\end{itemize}

\subsubsection{Khả sử dụng}
\begin{itemize}
    \item Giao diện người dùng hoàn toàn bằng tiếng Việt, phù hợp với đối tượng người dùng phổ thông tại Việt Nam.
    \item Luồng thao tác được thiết kế đơn giản: người dùng mới được dẫn qua onboarding từng bước; người dùng đã đăng nhập có thể thêm món ăn vào nhật ký chỉ với vài thao tác.
    \item Tính năng quét món ăn bằng ảnh giúp giảm thiểu thao tác nhập liệu thủ công.
    \item Giao diện sử dụng các thành phần Material Design quen thuộc, bố cục nhất quán giữa các màn hình.
\end{itemize}

\subsubsection{Ràng buộc kỹ thuật}
\begin{itemize}
    \item Ứng dụng di động hỗ trợ Android 5.0 (API 21) trở lên.
    \item Backend phụ thuộc hoàn toàn vào hệ sinh thái Firebase; không có máy chủ backend tự quản lý.
    \item Gemini API có giới hạn về số request mỗi phút và độ chính xác phụ thuộc vào chất lượng ảnh đầu vào và độ phủ của dữ liệu huấn luyện đối với ẩm thực Việt Nam.
    \item Tính năng quét AI và chatbot yêu cầu kết nối mạng để hoạt động.
\end{itemize}

            \item Gemini API trả về danh sách món ăn kèm thông tin dinh dưỡng ước tính (tên món, calo, protein, carbohydrate, chất béo, khối lượng phần ăn, độ tự tin) dưới dạng JSON.
            \item Cloud Function cập nhật số lượt quét đã sử dụng trong Firestore (đối với gói free), sau đó trả kết quả về ứng dụng.
            \item Ứng dụng hiển thị kết quả nhận diện cho người dùng kiểm tra.
            \item Người dùng chỉnh sửa thông tin dinh dưỡng nếu cần (tên món, khối lượng, calo,\ldots).
            \item Người dùng xác nhận, hệ thống lưu món ăn vào nhật ký (subcollection \texttt{meal\_entries}).
        \end{enumerate} \\
    \hline
    Luồng thay thế &
        \begin{enumerate}[nosep]
            \item[3a.] Ảnh không hợp lệ (dung lượng quá lớn, định dạng không hỗ trợ): Hiển thị thông báo ``Ảnh không hợp lệ, vui lòng chọn ảnh khác'' và cho phép chọn lại.
            \item[5a.] Hết quota quét AI (gói free): Cloud Function trả về lỗi quota\_exceeded. Ứng dụng hiển thị ``Bạn đã hết lượt quét miễn phí trong tháng này'' và gợi ý nâng cấp Premium hoặc nhập thủ công.
            \item[5b.] Gemini API không nhận diện được món ăn (confidence dưới ngưỡng): Cloud Function trả về danh sách rỗng. Ứng dụng hiển thị ``Không thể nhận diện món ăn, vui lòng thử lại với ảnh khác''.
            \item[5c.] Lỗi mạng hoặc Gemini API lỗi/quá tải: Cloud Function trả về lỗi. Ứng dụng hiển thị ``Đã xảy ra lỗi, vui lòng thử lại sau'' và cho phép người dùng thử lại hoặc nhập thủ công.
            \item[3b/8a.] Người dùng hủy thao tác: Quay về màn hình trước đó (màn hình chính hoặc màn hình nhật ký).
        \end{enumerate} \\
    \hline
    Tiền điều kiện & Người dùng đã đăng nhập; thiết bị có kết nối mạng. \\
    \hline
    Hậu điều kiện & Món ăn được lưu vào nhật ký dinh dưỡng của người dùng trong ngày hiện tại (subcollection \texttt{users/\{uid\}/meal\_entries}). Đối với gói free, số lượt quét AI trong tháng được tăng lên 1. \\
    \hline

        \begin{itemize}[nosep]

            \item Ảnh gửi lên phải có định dạng JPEG hoặc PNG, dung lượng không vượt quá giới hạn của Cloud Function request body.
        \end{itemize} \\

\section{Thiết kế cơ sở dữ liệu}

Hệ thống SmartNutri sử dụng \textbf{Cloud Firestore} — cơ sở dữ liệu NoSQL dạng document của Firebase. Khác với cơ sở dữ liệu quan hệ (RDBMS), Firestore không sử dụng bảng với schema cố định. Dữ liệu được tổ chức thành các \textbf{collection} (tập hợp document), mỗi \textbf{document} là một bản ghi JSON có cấu trúc linh hoạt. Một document có thể chứa các \textbf{sub-collection} — tập hợp document con nằm bên trong nó, tạo thành cấu trúc phân cấp.

\subsection{Về sơ đồ ERD}

Sơ đồ dưới đây (Hình~\ref{fig:erd_diagram}) mô tả \textbf{quan hệ logic} giữa các nhóm dữ liệu trong hệ thống. Do Firestore là NoSQL, sơ đồ này không biểu diễn quan hệ khóa ngoại như trong cơ sở dữ liệu quan hệ. Các mũi tên trong sơ đồ thể hiện mối liên kết dữ liệu được thực thi ở tầng ứng dụng hoặc thông qua Firestore Security Rules, không phải ràng buộc ở tầng cơ sở dữ liệu.

\subsection{Nguyên tắc phân tách dữ liệu người dùng}

Toàn bộ dữ liệu cá nhân của người dùng được tổ chức dưới đường dẫn \texttt{users/\{userId\}/}. Các sub-collection như \texttt{meal\_entries}, \texttt{favorites}, \texttt{usage}, \texttt{chat\_history} đều nằm trong document của từng người dùng. Firestore Security Rules đảm bảo mỗi người dùng chỉ có quyền đọc/ghi dữ liệu thuộc \texttt{users/\{userId\}/} của chính mình, trong đó \texttt{\{userId\}} được xác định từ Firebase Authentication token.
